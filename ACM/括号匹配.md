package com.company;

import java.util.*;

public class Main {
    public static void main(String[] args) {
        String str = "{}()[]";
        System.out.println(isRightStr(str));
    }

    /***
     * 题目：给定一个含有(){}[]三种括号的字符串，判断括号是否正确
     * 括号匹配算法：
     *		重点 1： 左括号必须使用同类型的右括号闭合
     *   特别重点 2： 括号必须以正确的顺序闭合（为了保证这点，只要右括号和栈顶元素匹配即可！）
     *   特别重点 3： 没必要匹配完整个字符串，一旦有问题可以直接判为 false，都判断完了没问题，则后返回true
     * @param str
     * @return
     */
    public static boolean isRightStr(String str) {
        //boolean isRight = false;
        Stack<Character> stack = new Stack<>();
        char[] strs = str.toCharArray();
        for (int i = 0; i < str.length(); i++) {
            //碰到左括号就入栈
            if ('(' == strs[i] || '{' == strs[i] || '[' == strs[i]) {
                stack.push(strs[i]);
            }
            //首先，碰到右括号应该先判断栈是否为空，为空说明直接错误
            //其次，栈不为空，再判断是否类型匹配
            if (')' == strs[i] || '}' == strs[i] || ']' == strs[i]) {
                if (stack.isEmpty())
                    return false;
                else {
                    //碰到右括号，那么就一定会出栈。不匹配就直接错误
                    char topchar = stack.pop();
                    if (')' == strs[i] && '(' == topchar)
                        return false;

                    if ('}' == strs[i] && '{' == topchar)
                        return false;

                    if (']' == strs[i] && '[' == topchar)
                        return false;
                }
            }
        }

        //return true;  //整个字符串检查完了都没返回false,不能说明是对的直接返回true。假如最后一个字符是左括号，这种情况就会误判
        return stack.isEmpty();     //整个字符串检查完了，并且栈为空，说明正确。
    }
}
