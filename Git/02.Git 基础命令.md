```java
git add				//1、添加新文件（或文件夹）到版本控制中
git add				//2、把修改过的文件放到暂存区，精确地将内容添加到下一次提交中。
```

```java
git commit -a 		//git add + git commit
git commit -m 'XXX'	 //提交更新并添加注释
```

```java
git status			//检查哪些文件处于什么状态
//使用 git status 提示：
Changes not staged for commit: 表示文件还没放到暂存区（not staged）    --此时需要add
Changes to be committed: 表示已有修改等着提交						--此时需要commit

git status -s 或者 git status -short 获得状态简览
//输出包括两个栏位，第一个字母栏位表示暂存区状态，第二个字母栏位表示工作区状态。
// A 表示新添加到暂存区的文件，M 表示修改过的文件
$ git status -s
?? LICENSE.txt			// ?? 标记文件新添加但还未跟踪
 M README				// README 文件在工作区，已修改但尚未暂存
A  lib/git.rb			// lib/git.rb 已暂存
M  lib/simplegit.rb		 // lib/simplegit.rb 文件已修改且已暂存
MM Rakefile				// Rakefile 文件已修改，暂存后又作了修改
```

```java
git ls-files		 //递归的查看当前目录及其子目录下的所有文件（在版本库中的）
git rm				//从版本库中移除文件（先查看文件在不在版本库）
// 要从 Git 中彻底移除某个文件，就必须要从已跟踪文件清单中移除，然后提交。
// 如果只是简单地从工作目录中手动删除文件，运行 git status 时就会在 “Changes not staged for commit” 部分看到：
$ git status
Changes not staged for commit:
		deleted:    PROJECTS.md
// 此时再运行 git rm 记录此次移除文件的操作：
$ git status
Changes to be committed:
		deleted:    PROJECTS.md
// 这样下次提交时，该文件彻底移除了
// 建议直接使用 GitHub Desktop 比较方便
```

```java
git diff			   //查看尚未暂存的文件更新了哪些部分
git diff --staged		//查看已经暂存的文件和最后一次提交的文件差异（等同于 git --cached ）
// 建议直接使用 GitHub Desktop
```

```java
git log				//查看提交历史，建议使用GitHub Desktop

// 其中一个比较有用的选项是 -p 或 -patch ，它会显示每次提交所引入的差异（按 补丁 的格式输出）
// 也可以限制显示日志条目的数量，例如使用 -2 选项来只显示最近的两次提交
git log -p -2

// 比如你想看到每次提交的 总览统计信息，可以使用 --stat 选项
git log --stat 

//定制化的输出想要的日志信息，可选参数online/short/full/fuller/format。详见官网
git log --pretty 
```
