//题目：实现删除字符串中出现次数最少的字符，若多个字符出现次数一样，则都删除。
//     输出删除这些单词后的字符串，字符串中其它字符保持原来的顺序。
        String str = sc.next();
        char[] chars = str.toCharArray();

        Map<Character, Integer> map = new LinkedHashMap<>();
        for (int i = 0; i < chars.length; i++) {
            if (!map.containsKey(chars[i]))
                map.put(chars[i], 1);
            else {
                int value = map.get(chars[i]);
                map.put(chars[i], ++value);  //当key值重复时，会自动覆盖value
            }
        }

        //map没有排序方法，对map排序前，先封装到List中。
        List<Map.Entry<Character, Integer>> list = new ArrayList<>(map.entrySet());
        //最缩减，最方便的写法：
        list.sort(Map.Entry.comparingByValue());
        
        
        //最原始的方法，调用Collections的sort方法，使用匿名内部类：
        //记住第二个参数 比较器接口：Comparator interface
        Collections.sort(list, new Comparator<Map.Entry<Character, Integer>>() {
            @Override
            public int compare(Map.Entry<Character, Integer> o1, Map.Entry<Character, Integer> o2) {
                return o1.getValue().compareTo(o2.getValue());
            }
        });
        
        //lambda表达式：
        Collections.sort(list, (o1, o2) -> o1.getValue().compareTo(o2.getValue()));
        
        
        //或者开始调用Map.Entry.comparingBy...
        Collections.sort(list, Map.Entry.comparingByValue());
        

        int min = list.get(0).getValue();

        for (Map.Entry<Character, Integer> characterIntegerEntry : list) {
            if (characterIntegerEntry.getValue() == min)
                str = str.replace(String.valueOf(characterIntegerEntry.getKey()), "");
        }
